====================================================================
GUÍA RÁPIDA: INSTALACIÓN, USO Y LIMPIEZA TOTAL (TERMINAL DIRECTO)
====================================================================

--------------------------------------------------------------------
PASO 1: INSTALAR Y CONECTAR CLOUDFLARE WARP 
--------------------------------------------------------------------

sudo ufw disable 2>/dev/null
curl -fsSL https://pkg.cloudflareclient.com/pubkey.gpg | sudo gpg --yes --dearmor --output /usr/share/keyrings/cloudflare-warp-archive-keyring.gpg
echo "deb [signed-by=/usr/share/keyrings/cloudflare-warp-archive-keyring.gpg] https://pkg.cloudflareclient.com/ $(lsb_release -cs) main" | sudo tee /etc/apt/sources.list.d/cloudflare-client.list
sudo apt update && sudo apt install -y cloudflare-warp
warp-cli registration new 2>/dev/null || warp-cli register
warp-cli connect
warp-cli status

--------------------------------------------------------------------
PASO 2: INSTALAR Y EJECUTAR CLI
--------------------------------------------------------------------

curl -fsSL https://antigravity.google/cli/install.sh | bash
export PATH="$HOME/.local/bin:$PATH"
agy

--------------------------------------------------------------------
PASO 3: DEJAR EL SISTEMA COMO ANTES
--------------------------------------------------------------------

# 1. Desconectar, borrar registro y purgar Cloudflare WARP
warp-cli disconnect 2>/dev/null
warp-cli registration delete 2>/dev/null || warp-cli delete 2>/dev/null
sudo apt purge -y cloudflare-warp
sudo apt autoremove -y
sudo rm -f /etc/apt/sources.list.d/cloudflare-client.list
sudo rm -f /usr/share/keyrings/cloudflare-warp-archive-keyring.gpg
rm -rf ~/.local/share/warp ~/.config/warp

# 2. Reactivar el Firewall (UFW)
sudo ufw enable

# 3. Eliminar Antigravity CLI, configuraciones, sesiones, cachés y credenciales
rm -f ~/.local/bin/agy
rm -rf ~/.gemini ~/.cache/antigravity

# 4. Limpiar variables en .bashrc, .zshrc y .profile
sed -i '/# Added by Antigravity CLI installer/d' ~/.bashrc ~/.zshrc ~/.profile 2>/dev/null
sed -i '\|export PATH=".*\.local/bin:\$PATH"|d' ~/.bashrc ~/.zshrc ~/.profile 2>/dev/null

echo "=== Sistema 100% restaurado y limpio ==="
